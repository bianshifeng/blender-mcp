# MetaOri 3D Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the MetaOri 3D module to the blender-mcp fork: text-to-image and image-to-3D on a local ComfyUI through swappable workflow packs, plus Blender-side import and normalization with UE FBX export, exposed as 7 MCP tools.

**Architecture:** A new package `src/blender_mcp/metaori/` inside the MCP server process. It talks to ComfyUI directly over HTTP as async jobs (submit / poll / download) and only sends bpy scripts to Blender through the addon's existing `execute_code` command. `server.py` gains three lines; `addon.py` is untouched.

**Tech Stack:** Python ≥3.10, `mcp` FastMCP 1.x, `httpx` (async), pytest (run via `uv run --with pytest`), Blender 5.1 `bpy`/`bmesh`/`mathutils`/`numpy` for the bpy scripts.

**Spec:** `docs/superpowers/specs/2026-09-28-metaori-3d-design.md`

## Global Constraints

- No new runtime dependencies: only `mcp>=1.9.0,<2` and `httpx>=0.27.0` from `pyproject.toml`.
- Do not modify `addon.py` or `src/blender_mcp/bundled/addon.py`.
- `server.py` changes are limited to registering MetaOri and one instructions line; `SERVER_INSTRUCTIONS` must stay `< 2500` characters.
- Identifiers, comments, log and error messages in English.
- `asset_name` must match `^[A-Za-z][A-Za-z0-9]*(?:_[A-Za-z0-9]+)*$` and be at most 64 characters.
- Job folders `<METAORI_OUTPUT_DIR>/<AssetName>/<YYYYMMDD-HHMMSS>_<t2i|i2m>_<job8>/` are never overwritten; an existing FBX export is only replaced with `overwrite_export=true`.
- Environment: `METAORI_COMFYUI_URL` (default `http://127.0.0.1:8188`), `METAORI_OUTPUT_DIR` (default `~/MetaOri/AI_Raw`), `METAORI_EXPORT_DIR` (default `~/MetaOri/Export`); relative paths resolve against the process working directory.
- UE FBX preset, verbatim: `object_types={'MESH'}, global_scale=1.0, apply_unit_scale=True, apply_scale_options='FBX_SCALE_NONE', axis_forward='-Z', axis_up='Y', mesh_smooth_type='FACE', use_mesh_modifiers=True, add_leaf_bones=False, bake_anim=False`.
- bpy scripts use only `bpy`, `bmesh`, `mathutils`, `numpy` and the standard library, and must run on Blender 5.1.
- Test command (from `Tools/blender-mcp`): `uv run --frozen --with pytest pytest -q`. Baseline before this work: 269 passed, 1 skipped.

```python
"""Exceptions raised by MetaOri. The MCP layer reports them as tool errors."""


class MetaOriError(Exception):
    """Base class for every MetaOri failure."""


class ComfyUnavailable(MetaOriError):
    """ComfyUI could not be reached."""


class ComfyRequestError(MetaOriError):
    """ComfyUI answered with an error, such as a workflow validation failure."""

    def __init__(self, message, details=None):
        super().__init__(message)
        self.details = details or {}


class WorkflowError(MetaOriError):
    """A workflow pack is invalid, missing, or cannot serve a request."""


class ParamError(MetaOriError):
    """A tool argument is invalid."""


class JobNotFound(MetaOriError):
    """No job record matches the given id."""


class BlenderScriptError(MetaOriError):
    """A MetaOri bpy script failed inside Blender."""
```

```python
"""MetaOri configuration, read from environment variables."""
from __future__ import annotations

import os
from collections.abc import Mapping
from dataclasses import dataclass
from pathlib import Path

DEFAULT_COMFYUI_URL = "http://127.0.0.1:8188"


@dataclass(frozen=True)
class MetaOriConfig:
    comfyui_url: str
    output_dir: Path
    export_dir: Path

    @classmethod
    def from_env(cls, env: Mapping[str, str] | None = None, cwd: Path | None = None) -> "MetaOriConfig":
        """Build the config; relative directories resolve against cwd (default: process cwd)."""
        env = os.environ if env is None else env
        base = Path(cwd) if cwd is not None else Path.cwd()

        def directory(name: str, default: Path) -> Path:
            raw = (env.get(name) or "").strip()
            path = Path(raw).expanduser() if raw else default
            return (path if path.is_absolute() else base / path).resolve()

        url = (env.get("METAORI_COMFYUI_URL") or "").strip() or DEFAULT_COMFYUI_URL
        home = Path.home() / "MetaOri"
        return cls(
            comfyui_url=url.rstrip("/"),
            output_dir=directory("METAORI_OUTPUT_DIR", home / "AI_Raw"),
            export_dir=directory("METAORI_EXPORT_DIR", home / "Export"),
        )
```

```python
"""Capability contracts: the standard inputs and outputs every workflow of a capability implements.

Also holds the asset-name rule and the purpose presets that shape text-to-image prompts.
"""
from __future__ import annotations

import json
import re
from dataclasses import dataclass
from pathlib import Path
from typing import Any

from .errors import ParamError

ASSET_NAME_RE = re.compile(r"^[A-Za-z][A-Za-z0-9]*(?:_[A-Za-z0-9]+)*$")
ASSET_NAME_MAX = 64
MAX_SEED = 2**53 - 1
PRESETS_PATH = Path(__file__).with_name("presets.json")


@dataclass(frozen=True)
class ParamSpec:
    name: str
    type: type
    description: str
    min: int | None = None
    max: int | None = None
    multiple_of: int | None = None
    choices: tuple[int, ...] | None = None
    max_length: int | None = None

    def validate(self, value: Any) -> Any:
        if self.type is bool:
            if not isinstance(value, bool):
                raise ParamError(f"{self.name} must be true or false, got {value!r}")
            return value
        if self.type is str:
            if not isinstance(value, str):
                raise ParamError(f"{self.name} must be a string, got {value!r}")
            if self.max_length is not None and len(value) > self.max_length:
                raise ParamError(f"{self.name} is longer than {self.max_length} characters")
            return value
        if isinstance(value, bool) or not isinstance(value, int):
            raise ParamError(f"{self.name} must be an integer, got {value!r}")
        if self.min is not None and value < self.min:
            raise ParamError(f"{self.name} must be >= {self.min}, got {value}")
        if self.max is not None and value > self.max:
            raise ParamError(f"{self.name} must be <= {self.max}, got {value}")
        if self.multiple_of and value % self.multiple_of:
            raise ParamError(f"{self.name} must be a multiple of {self.multiple_of}, got {value}")
        if self.choices is not None and value not in self.choices:
            raise ParamError(f"{self.name} must be one of {list(self.choices)}, got {value}")
        return value


@dataclass(frozen=True)
class Capability:
    name: str
    kind: str
    inputs: dict[str, ParamSpec]
    required_inputs: tuple[str, ...]
    outputs: tuple[str, ...]
    internal_inputs: tuple[str, ...] = ()

    def validate(self, params: dict[str, Any]) -> dict[str, Any]:
        """Check standard parameters; None values are dropped (meaning "not given")."""
        unknown = sorted(set(params) - set(self.inputs))
        if unknown:
            raise ParamError(f"unknown {self.name} parameters: {unknown}")
        return {key: self.inputs[key].validate(value) for key, value in params.items() if value is not None}


def _seed() -> ParamSpec:
    return ParamSpec("seed", int, "Sampler seed; random when omitted", min=0, max=MAX_SEED)


def _prefix() -> ParamSpec:
    return ParamSpec("filename_prefix", str, "ComfyUI output prefix, set by MetaOri", max_length=200)


TEXT_TO_IMAGE = Capability(
    name="text_to_image",
    kind="t2i",
    inputs={
        "prompt": ParamSpec("prompt", str, "Positive prompt", max_length=4000),
        "negative_prompt": ParamSpec("negative_prompt", str, "Negative prompt", max_length=4000),
        "width": ParamSpec("width", int, "Image width in pixels", min=256, max=2048, multiple_of=8),
        "height": ParamSpec("height", int, "Image height in pixels", min=256, max=2048, multiple_of=8),
        "seed": _seed(),
        "count": ParamSpec("count", int, "Images per job", min=1, max=4),
        "filename_prefix": _prefix(),
    },
    required_inputs=("prompt",),
    outputs=("images",),
    internal_inputs=("filename_prefix",),
)

IMAGE_TO_3D = Capability(
    name="image_to_3d",
    kind="i2m",
    inputs={
        "image": ParamSpec("image", str, "ComfyUI input image name, set by MetaOri after upload", max_length=500),
        "seed": _seed(),
        "target_faces": ParamSpec("target_faces", int, "Face count after decimation", min=1000, max=2_000_000),
        "texture_size": ParamSpec("texture_size", int, "Baked texture resolution", choices=(512, 1024, 2048, 4096)),
        "remove_background": ParamSpec("remove_background", bool, "Remove the image background first"),
        "filename_prefix": _prefix(),
    },
    required_inputs=("image",),
    outputs=("model",),
    internal_inputs=("image", "filename_prefix"),
)

CAPABILITIES = {capability.name: capability for capability in (TEXT_TO_IMAGE, IMAGE_TO_3D)}


def validate_asset_name(name: Any) -> str:
    if not isinstance(name, str) or len(name) > ASSET_NAME_MAX or not ASSET_NAME_RE.match(name):
        raise ParamError(
            "asset_name must be PascalCase letters and digits with optional _Variant parts "
            f"(e.g. BronzeCenser or WoodCrate_A), at most {ASSET_NAME_MAX} characters; got {name!r}"
        )
    return name


@dataclass(frozen=True)
class PromptPreset:
    name: str
    description: str
    positive_suffix: str
    negative: str
    width: int | None
    height: int | None


@dataclass(frozen=True)
class ComposedPrompt:
    positive: str
    negative: str | None  # None leaves the workflow's own negative prompt in place


def load_presets(path: Path = PRESETS_PATH) -> dict[str, PromptPreset]:
    data = json.loads(Path(path).read_text(encoding="utf-8"))
    return {
        name: PromptPreset(
            name=name,
            description=spec.get("description", ""),
            positive_suffix=spec.get("positive_suffix", ""),
            negative=spec.get("negative", ""),
            width=spec.get("width"),
            height=spec.get("height"),
        )
        for name, spec in data["purposes"].items()
    }


def compose_prompt(preset: PromptPreset, prompt: str, negative_prompt: str | None = None) -> ComposedPrompt:
    """Append the preset to the caller's prompt; the negative is preset + caller, comma-joined."""
    text = (prompt or "").strip()
    if not text:
        raise ParamError("prompt must not be empty")
    positive = f"{text}, {preset.positive_suffix}" if preset.positive_suffix else text
    parts = [part.strip() for part in (preset.negative, negative_prompt or "") if part and part.strip()]
    return ComposedPrompt(positive=positive, negative=", ".join(parts) if parts else None)
```

```python
"""Workflow packs: an untouched ComfyUI API graph plus a manifest binding standard parameters to nodes.

To swap in a better workflow, export it from ComfyUI with "Export (API)", save it as
workflows/<id>/workflow_api.json and describe it in workflows/<id>/manifest.json
(see docs/metaori.md). No code changes are needed.
"""
from __future__ import annotations

import copy
import hashlib
import json
from dataclasses import dataclass
from pathlib import Path
from typing import Any

from .capabilities import CAPABILITIES, Capability
from .errors import ParamError, WorkflowError

MANIFEST_SCHEMA = "metaori.workflow/1"
WORKFLOWS_DIR = Path(__file__).with_name("workflows")
EXTRA_TYPES = ("bool", "enum", "float", "int", "str")


@dataclass(frozen=True)
class Target:
    node: str
    class_type: str
    input: str


@dataclass(frozen=True)
class Extra:
    name: str
    type: str
    description: str
    default: Any
    targets: tuple[Target, ...]
    min: float | None = None
    max: float | None = None
    values: dict[str, Any] | None = None

    def convert(self, value: Any) -> Any:
        """Validate a caller value and return the raw value written into the graph."""
        if self.type == "enum":
            if not isinstance(value, str) or value not in self.values:
                raise ParamError(f"{self.name} must be one of {sorted(self.values)}, got {value!r}")
            return self.values[value]
        if self.type == "bool":
            if not isinstance(value, bool):
                raise ParamError(f"{self.name} must be true or false, got {value!r}")
            return value
        if self.type == "str":
            if not isinstance(value, str):
                raise ParamError(f"{self.name} must be a string, got {value!r}")
            return value
        if self.type == "int" and (isinstance(value, bool) or not isinstance(value, int)):
            raise ParamError(f"{self.name} must be an integer, got {value!r}")
        if self.type == "float":
            if isinstance(value, bool) or not isinstance(value, (int, float)):
                raise ParamError(f"{self.name} must be a number, got {value!r}")
            value = float(value)
        if self.min is not None and value < self.min:
            raise ParamError(f"{self.name} must be >= {self.min}, got {value}")
        if self.max is not None and value > self.max:
            raise ParamError(f"{self.name} must be <= {self.max}, got {value}")
        return value

    def describe(self) -> dict[str, Any]:
        info: dict[str, Any] = {"type": self.type, "description": self.description, "default": self.default}
        if self.values is not None:
            info["values"] = sorted(self.values)
        if self.min is not None:
            info["min"] = self.min
        if self.max is not None:
            info["max"] = self.max
        return info


@dataclass(frozen=True)
class Output:
    name: str
    node: str
    class_type: str
    key: str


@dataclass
class WorkflowPack:
    id: str
    capability: Capability
    description: str
    default: bool
    graph: dict[str, Any]
    graph_sha256: str
    inputs: dict[str, tuple[Target, ...]]
    extras: dict[str, Extra]
    outputs: dict[str, Output]
    directory: Path

    def supports(self, param: str) -> bool:
        return param in self.inputs

    def convert_extras(self, extras: dict[str, Any] | None) -> dict[str, Any]:
        """Validate caller extras; returns name -> raw graph value."""
        extras = extras or {}
        unknown = sorted(set(extras) - set(self.extras))
        if unknown:
            raise ParamError(
                f"workflow {self.id} has no extra parameters {unknown}; available: {sorted(self.extras)}")
        return {name: self.extras[name].convert(value) for name, value in extras.items()}

    def bind(self, params: dict[str, Any], extras: dict[str, Any] | None = None) -> dict[str, Any]:
        """Return a copy of the graph with standard params and extras written into their targets.

        A target that currently holds a link ([node_id, output_index]) is replaced by the constant.
        """
        unsupported = sorted(set(params) - set(self.inputs))
        if unsupported:
            raise WorkflowError(
                f"workflow {self.id} does not support {unsupported}; its inputs are {sorted(self.inputs)}")
        raw_extras = self.convert_extras(extras)
        graph = copy.deepcopy(self.graph)
        for name, value in params.items():
            for target in self.inputs[name]:
                graph[target.node]["inputs"][target.input] = value
        for name, value in raw_extras.items():
            for target in self.extras[name].targets:
                graph[target.node]["inputs"][target.input] = value
        return graph

    def resolve_extras(self, extras: dict[str, Any] | None) -> dict[str, Any]:
        """Caller extras merged over the manifest defaults, as names (for the job record)."""
        merged = {name: extra.default for name, extra in self.extras.items()}
        merged.update(extras or {})
        return merged

    def required_classes(self) -> set[str]:
        return {node["class_type"] for node in self.graph.values()}

    def describe(self) -> dict[str, Any]:
        return {
            "id": self.id,
            "capability": self.capability.name,
            "default": self.default,
            "description": self.description,
            "inputs": sorted(k for k in self.inputs if k not in self.capability.internal_inputs),
            "extras": {name: extra.describe() for name, extra in sorted(self.extras.items())},
            "outputs": sorted(self.outputs),
        }


def _targets(raw: Any, graph: dict[str, Any], where: str) -> tuple[Target, ...]:
    if not isinstance(raw, list) or not raw:
        raise WorkflowError(f"{where}: bind a non-empty list of targets")
    targets = []
    for item in raw:
        try:
            node, class_type, field = str(item["node"]), item["class_type"], item["input"]
        except (KeyError, TypeError):
            raise WorkflowError(f"{where}: every target needs node, class_type and input") from None
        if node not in graph:
            raise WorkflowError(f"{where}: node {node} not in graph")
        actual = graph[node].get("class_type")
        if actual != class_type:
            raise WorkflowError(f"{where}: node {node} is {actual}, manifest says {class_type}")
        if field not in graph[node].get("inputs", {}):
            raise WorkflowError(f"{where}: node {node} ({class_type}) has no input {field!r}")
        targets.append(Target(node, class_type, field))
    return tuple(targets)


def _read_json(path: Path, what: str) -> tuple[Any, bytes]:
    try:
        data = path.read_bytes()
    except FileNotFoundError:
        raise WorkflowError(f"{what} missing: {path.name}") from None
    try:
        return json.loads(data.decode("utf-8")), data
    except (UnicodeDecodeError, json.JSONDecodeError) as exc:
        raise WorkflowError(f"{what} is not valid JSON: {exc}") from None


def _extra(pack_id: str, capability: Capability, name: str, raw: Any, graph: dict[str, Any]) -> Extra:
    where = f"{pack_id} extras.{name}"
    if name in capability.inputs:
        raise WorkflowError(f"{where}: clashes with the standard input of the same name")
    if not isinstance(raw, dict) or raw.get("type") not in EXTRA_TYPES:
        raise WorkflowError(f"{where}: needs a type in {list(EXTRA_TYPES)}")
    values = raw.get("values")
    if raw["type"] == "enum" and (not isinstance(values, dict) or not values):
        raise WorkflowError(f"{where}: an enum needs a non-empty values map")
    extra = Extra(
        name=name, type=raw["type"], description=raw.get("description", ""), default=raw.get("default"),
        targets=_targets(raw.get("bind"), graph, where), min=raw.get("min"), max=raw.get("max"),
        values=values if raw["type"] == "enum" else None,
    )
    try:
        default_raw = extra.convert(extra.default)
    except ParamError as exc:
        raise WorkflowError(f"{where}: default is invalid: {exc}") from None
    for target in extra.targets:
        current = graph[target.node]["inputs"][target.input]
        if not isinstance(current, list) and current != default_raw:
            raise WorkflowError(
                f"{where}: default {extra.default!r} does not match node {target.node}.{target.input} = {current!r}")
    return extra


def load_pack(directory: Path) -> WorkflowPack:
    directory = Path(directory)
    manifest, _ = _read_json(directory / "manifest.json", f"{directory.name}: manifest")
    if not isinstance(manifest, dict) or manifest.get("schema") != MANIFEST_SCHEMA:
        raise WorkflowError(f"{directory.name}: manifest schema must be {MANIFEST_SCHEMA!r}")
    pack_id = manifest.get("id")
    if pack_id != directory.name:
        raise WorkflowError(f"{directory.name}: manifest id {pack_id!r} must match the folder name")
    capability = CAPABILITIES.get(manifest.get("capability"))
    if capability is None:
        raise WorkflowError(f"{pack_id}: unknown capability {manifest.get('capability')!r}")
    graph, graph_bytes = _read_json(directory / manifest.get("graph", "workflow_api.json"), f"{pack_id}: graph")
    if not isinstance(graph, dict) or not graph or not all(
            isinstance(node, dict) and "class_type" in node for node in graph.values()):
        raise WorkflowError(f"{pack_id}: graph is not a ComfyUI API-format graph (use 'Export (API)' in ComfyUI)")

    inputs = {}
    for name, raw in (manifest.get("inputs") or {}).items():
        if name not in capability.inputs:
            raise WorkflowError(
                f"{pack_id}: {name!r} is not a standard {capability.name} input; declare it under extras")
        inputs[name] = _targets(raw, graph, f"{pack_id} inputs.{name}")
    missing = [name for name in capability.required_inputs if name not in inputs]
    if missing:
        raise WorkflowError(f"{pack_id}: required inputs not bound: {missing}")

    extras = {name: _extra(pack_id, capability, name, raw, graph)
              for name, raw in (manifest.get("extras") or {}).items()}

    outputs = {}
    for name, raw in (manifest.get("outputs") or {}).items():
        if name not in capability.outputs:
            raise WorkflowError(f"{pack_id}: {name!r} is not a standard {capability.name} output")
        try:
            node, class_type, key = str(raw["node"]), raw["class_type"], raw["key"]
        except (KeyError, TypeError):
            raise WorkflowError(f"{pack_id} outputs.{name}: needs node, class_type and key") from None
        if node not in graph:
            raise WorkflowError(f"{pack_id} outputs.{name}: node {node} not in graph")
        if graph[node].get("class_type") != class_type:
            raise WorkflowError(
                f"{pack_id} outputs.{name}: node {node} is {graph[node].get('class_type')}, manifest says {class_type}")
        outputs[name] = Output(name, node, class_type, key)
    missing = [name for name in capability.outputs if name not in outputs]
    if missing:
        raise WorkflowError(f"{pack_id}: outputs not declared: {missing}")

    return WorkflowPack(
        id=pack_id, capability=capability, description=manifest.get("description", ""),
        default=bool(manifest.get("default", False)), graph=graph,
        graph_sha256=hashlib.sha256(graph_bytes).hexdigest(), inputs=inputs, extras=extras,
        outputs=outputs, directory=directory,
    )


class WorkflowRegistry:
    """All workflow packs in a folder; packs that fail validation are listed in `errors`."""

    def __init__(self, packs: dict[str, WorkflowPack], errors: dict[str, str]):
        self.packs = packs
        self.errors = errors

    @classmethod
    def load(cls, root: Path = WORKFLOWS_DIR) -> "WorkflowRegistry":
        packs: dict[str, WorkflowPack] = {}
        errors: dict[str, str] = {}
        for directory in sorted(path for path in Path(root).iterdir() if path.is_dir()):
            if directory.name.startswith((".", "_")):
                continue
            try:
                pack = load_pack(directory)
            except WorkflowError as exc:
                errors[directory.name] = str(exc)
                continue
            packs[pack.id] = pack
        for capability in CAPABILITIES:
            defaults = sorted(p.id for p in packs.values() if p.capability.name == capability and p.default)
            if len(defaults) > 1:
                for pack_id in defaults:
                    errors[pack_id] = f"more than one default {capability} workflow: {defaults}"
                    del packs[pack_id]
        return cls(packs, errors)

    def get(self, capability: str, workflow_id: str | None = None) -> WorkflowPack:
        if workflow_id:
            pack = self.packs.get(workflow_id)
            if pack is None:
                reason = self.errors.get(workflow_id)
                detail = f": {reason}" if reason else f"; available: {sorted(self.packs)}"
                raise WorkflowError(f"workflow {workflow_id!r} is not available{detail}")
            if pack.capability.name != capability:
                raise WorkflowError(f"workflow {workflow_id!r} implements {pack.capability.name}, not {capability}")
            return pack
        candidates = sorted((p for p in self.packs.values() if p.capability.name == capability), key=lambda p: p.id)
        if not candidates:
            raise WorkflowError(f"no {capability} workflow is available")
        return next((p for p in candidates if p.default), candidates[0])
```

```python
"""Async client for the parts of the ComfyUI server API that MetaOri uses."""
from __future__ import annotations

import hashlib
import os
import uuid
from dataclasses import dataclass
from pathlib import Path
from typing import Any

import httpx

from .errors import ComfyRequestError, ComfyUnavailable

UPLOAD_SUBFOLDER = "metaori"
DOWNLOAD_TIMEOUT = httpx.Timeout(30.0, read=300.0)


@dataclass(frozen=True)
class QueueState:
    running: list[str]
    pending: list[str]  # in execution order

    def position(self, prompt_id: str) -> int | None:
        """1-based place among pending prompts, or None when not pending."""
        return self.pending.index(prompt_id) + 1 if prompt_id in self.pending else None


class ComfyClient:
    def __init__(self, base_url: str, *, transport: httpx.AsyncBaseTransport | None = None, timeout: float = 30.0):
        self.base_url = base_url.rstrip("/")
        self.client_id = uuid.uuid4().hex
        self._transport = transport
        self._timeout = timeout

    def _client(self) -> httpx.AsyncClient:
        return httpx.AsyncClient(base_url=self.base_url, transport=self._transport, timeout=self._timeout)

    def _unavailable(self, exc: Exception) -> ComfyUnavailable:
        return ComfyUnavailable(
            f"ComfyUI is not reachable at {self.base_url} ({type(exc).__name__}: {exc}). "
            "Start ComfyUI or fix METAORI_COMFYUI_URL.")

    async def _request(self, method: str, path: str, **kwargs: Any) -> httpx.Response:
        try:
            async with self._client() as client:
                return await client.request(method, path, **kwargs)
        except httpx.TransportError as exc:
            raise self._unavailable(exc) from exc

    async def _json(self, method: str, path: str, **kwargs: Any) -> Any:
        response = await self._request(method, path, **kwargs)
        if response.status_code != 200:
            raise ComfyRequestError(
                f"ComfyUI {method} {path} failed with HTTP {response.status_code}: {response.text[:500]}")
        return response.json()

    async def _expect_ok(self, method: str, path: str, **kwargs: Any) -> None:
        response = await self._request(method, path, **kwargs)
        if response.status_code != 200:
            raise ComfyRequestError(f"ComfyUI {method} {path} failed with HTTP {response.status_code}")

    async def system_stats(self) -> dict[str, Any]:
        return await self._json("GET", "/system_stats")

    async def object_info(self) -> dict[str, Any]:
        return await self._json("GET", "/object_info", timeout=120.0)

    async def upload_image(self, path: Path) -> str:
        """Upload into ComfyUI's input/metaori folder; returns the value for a LoadImage node."""
        data = Path(path).read_bytes()
        name = hashlib.sha256(data).hexdigest()[:16] + Path(path).suffix.lower()
        response = await self._request(
            "POST", "/upload/image",
            files={"image": (name, data, "application/octet-stream")},
            data={"type": "input", "subfolder": UPLOAD_SUBFOLDER, "overwrite": "true"},
        )
        if response.status_code != 200:
            raise ComfyRequestError(f"image upload failed with HTTP {response.status_code}: {response.text[:500]}")
        body = response.json()
        subfolder = body.get("subfolder") or ""
        return f"{subfolder}/{body['name']}" if subfolder else body["name"]

    async def queue_prompt(self, graph: dict[str, Any], prompt_id: str) -> None:
        response = await self._request(
            "POST", "/prompt", json={"prompt": graph, "prompt_id": prompt_id, "client_id": self.client_id})
        if response.status_code == 200:
            return
        try:
            body = response.json()
        except ValueError:
            body = {"error": {"message": response.text[:500]}}
        error = body.get("error") or {}
        message = error.get("message") if isinstance(error, dict) else str(error)
        raise ComfyRequestError(f"ComfyUI rejected the workflow: {message}",
                                details={"error": error, "node_errors": body.get("node_errors") or {}})

    async def history(self, prompt_id: str) -> dict[str, Any] | None:
        body = await self._json("GET", f"/history/{prompt_id}")
        return body.get(prompt_id)

    async def queue(self) -> QueueState:
        body = await self._json("GET", "/queue")

        def ids(items: list) -> list[str]:
            return [item[1] for item in sorted(items, key=lambda item: item[0])]

        return QueueState(running=ids(body.get("queue_running") or []), pending=ids(body.get("queue_pending") or []))

    async def download(self, filename: str, subfolder: str, folder_type: str, dest: Path) -> Path:
        """Stream one output file to dest through a .part file, so a failed download leaves nothing."""
        dest = Path(dest)
        dest.parent.mkdir(parents=True, exist_ok=True)
        part = dest.with_name(dest.name + ".part")
        params = {"filename": filename, "subfolder": subfolder, "type": folder_type}
        try:
            async with self._client() as client:
                async with client.stream("GET", "/view", params=params, timeout=DOWNLOAD_TIMEOUT) as response:
                    if response.status_code != 200:
                        raise ComfyRequestError(
                            f"download of {subfolder}/{filename} failed with HTTP {response.status_code}")
                    with open(part, "wb") as handle:
                        async for chunk in response.aiter_bytes():
                            handle.write(chunk)
        except httpx.TransportError as exc:
            part.unlink(missing_ok=True)
            raise self._unavailable(exc) from exc
        except BaseException:
            part.unlink(missing_ok=True)
            raise
        os.replace(part, dest)
        return dest

    async def delete_pending(self, prompt_id: str) -> None:
        await self._expect_ok("POST", "/queue", json={"delete": [prompt_id]})

    async def interrupt(self, prompt_id: str) -> None:
        await self._expect_ok("POST", "/interrupt", json={"prompt_id": prompt_id})
```

```python
"""Job records: one folder per job under <output_dir>/<AssetName>/, never overwritten.

The folder keeps the downloaded outputs next to job.json, which records the prompt, seed,
parameters and workflow version, so every AI-generated asset stays traceable.
"""
from __future__ import annotations

import json
import os
import uuid
from datetime import datetime
from pathlib import Path
from typing import Any

from .errors import JobNotFound, MetaOriError

JOB_SCHEMA = "metaori.job/1"
JOB_FILE = "job.json"
TERMINAL = frozenset({"succeeded", "failed", "cancelled"})


def new_job_id() -> str:
    return str(uuid.uuid4())


def now_iso() -> str:
    return datetime.now().astimezone().isoformat(timespec="seconds")


def _timestamp() -> str:
    return datetime.now().strftime("%Y%m%d-%H%M%S")


class JobStore:
    def __init__(self, output_dir: Path):
        self.output_dir = Path(output_dir)
        self._index: dict[str, Path] = {}

    def create(self, *, job_id: str, kind: str, asset_name: str,
               record: dict[str, Any]) -> tuple[Path, dict[str, Any]]:
        folder = self.output_dir / asset_name / f"{_timestamp()}_{kind}_{job_id[:8]}"
        try:
            folder.mkdir(parents=True, exist_ok=False)
        except FileExistsError:
            raise MetaOriError(f"job folder {folder} already exists; refusing to overwrite it") from None
        full = {"schema": JOB_SCHEMA, "job_id": job_id, "asset_name": asset_name, "folder": str(folder),
                "created_at": now_iso(), **record}
        self.save(full)
        return folder, full

    def save(self, record: dict[str, Any]) -> None:
        folder = Path(record["folder"])
        part = folder / (JOB_FILE + ".part")
        part.write_text(json.dumps(record, ensure_ascii=False, indent=2), encoding="utf-8")
        os.replace(part, folder / JOB_FILE)
        self._index[record["job_id"]] = folder

    def load(self, job_id: str) -> dict[str, Any]:
        try:
            canonical = str(uuid.UUID(str(job_id)))
        except ValueError:
            raise JobNotFound(f"{job_id!r} is not a MetaOri job id") from None
        folder = self._index.get(canonical) or self._scan(canonical)
        if folder is None:
            raise JobNotFound(f"no MetaOri job {canonical} under {self.output_dir}")
        record = json.loads((folder / JOB_FILE).read_text(encoding="utf-8"))
        self._index[canonical] = folder
        return record

    def _scan(self, job_id: str) -> Path | None:
        if not self.output_dir.is_dir():
            return None
        for path in self.output_dir.glob(f"*/*_{job_id[:8]}/{JOB_FILE}"):
            try:
                if json.loads(path.read_text(encoding="utf-8")).get("job_id") == job_id:
                    return path.parent
            except (OSError, ValueError):
                continue
        return None
```

```python
            raise ComfyUnavailable("ComfyUI is not reachable at http://comfy.test")

    async def system_stats(self):
        self._check()
        return {"system": {"comfyui_version": "0.37.0"},
                "devices": [{"name": "cuda:0 RTX 4090", "vram_total": 24 * 2**30, "vram_free": 8 * 2**30}]}

    async def object_info(self):
        self._check()
        return {name: {} for name in self.object_info_classes}

    async def upload_image(self, path):
        self._check()
        self.uploads.append(Path(path))
        return "metaori/abc.png"

    async def queue_prompt(self, graph, prompt_id):
        self._check()
        if self.submit_error:
            raise self.submit_error
        self.submitted.append((prompt_id, graph))
        self.pending.append(prompt_id)

    async def history(self, prompt_id):
        self._check()
        return self.histories.get(prompt_id)

    async def queue(self):
        self._check()
        return QueueState(running=list(self.running), pending=list(self.pending))

    async def download(self, filename, subfolder, folder_type, dest):
        self._check()
        dest = Path(dest)
        dest.parent.mkdir(parents=True, exist_ok=True)
        dest.write_bytes(self.files[(subfolder, filename)])
        self.downloads.append(dest)
        return dest

    async def delete_pending(self, prompt_id):
        self.deleted.append(prompt_id)
        self.pending.remove(prompt_id)

    async def interrupt(self, prompt_id):
        self.interrupted.append(prompt_id)

    def start(self, prompt_id):
        self.pending.remove(prompt_id)
        self.running.append(prompt_id)

    def _leave_queue(self, prompt_id):
        for queue in (self.pending, self.running):
            if prompt_id in queue:
                queue.remove(prompt_id)

    def succeed(self, prompt_id, node, key, files):
        """files: list of (subfolder, filename, bytes) the output node reports."""
        self._leave_queue(prompt_id)
        items = []
        for subfolder, filename, data in files:
            self.files[(subfolder, filename)] = data
            items.append({"filename": filename, "subfolder": subfolder, "type": "output"})
        self.histories[prompt_id] = {
            "status": {"status_str": "success", "completed": True, "messages": [
                ["execution_start", {"timestamp": 1_000}], ["execution_success", {"timestamp": 13_500}]]},
            "outputs": {node: {key: items}},
        }

    def fail(self, prompt_id, message="CUDA out of memory"):
        self._leave_queue(prompt_id)
        self.histories[prompt_id] = {
            "status": {"status_str": "error", "completed": False, "messages": [
                ["execution_start", {"timestamp": 1_000}],
                ["execution_error", {"node_id": "458", "node_type": "KSampler", "exception_message": message,

            "outputs": {},
        }

    def interrupt_history(self, prompt_id):
        self._leave_queue(prompt_id)
        self.histories[prompt_id] = {
            "status": {"status_str": "error", "completed": False, "messages": [
                ["execution_start", {"timestamp": 1_000}], ["execution_interrupted", {"timestamp": 1_500}]]},
            "outputs": {},
        }


class FakeClock:
    """Monotonic clock plus an async sleep that advances it; on_sleep lets a test act mid-wait."""

    def __init__(self):
        self.now = 0.0
        self.sleeps: list[float] = []
        self.on_sleep = None

    def __call__(self):
        return self.now

    async def sleep(self, seconds):
        self.sleeps.append(seconds)
        self.now += seconds
        if self.on_sleep:
            self.on_sleep()
```

```python
"""MetaOri orchestration: validate, bind, submit, poll and download.

The MCP tools in tools.py are thin wrappers around MetaOriService. It only needs a ComfyUI
client, the workflow registry and a job store, so it is tested against fakes.

from __future__ import annotations

import asyncio
import random

import time
from datetime import datetime

from typing import Any, Awaitable, Callable

from .capabilities import IMAGE_TO_3D, MAX_SEED, TEXT_TO_IMAGE, compose_prompt, load_presets, validate_asset_name

POLL_INTERVAL = 2.0
MAX_WAIT_SECONDS = 600.0
OBJECT_INFO_TTL = 300.0
IMAGE_SUFFIXES = (".png", ".jpg", ".jpeg", ".webp")
FINISH_EVENTS = ("execution_success", "execution_error", "execution_interrupted")


def random_seed() -> int:
    return random.randint(0, MAX_SEED)


def _output_name(name: str, index: int, total: int, source_filename: str) -> str:
    suffix = Path(source_filename).suffix.lower() or ".bin"
    if name == "images":
        return f"image_{index:02d}{suffix}"
    return f"{name}{suffix}" if total == 1 else f"{name}_{index:02d}{suffix}"


def _execution_seconds(events: dict[str, dict]) -> float | None:
    start = (events.get("execution_start") or {}).get("timestamp")
    end = next((events[key].get("timestamp") for key in FINISH_EVENTS if key in events), None)
    if start is None or end is None:
        return None
    return round((end - start) / 1000.0, 1)


def _seconds_between(start: str | None, end: str) -> float | None:
    try:
        return round((datetime.fromisoformat(end) - datetime.fromisoformat(start)).total_seconds(), 1)
    except (TypeError, ValueError):
        return None


class MetaOriService:
    """Everything the metaori_* tools do, minus MCP and Blender."""

    def __init__(self, config: MetaOriConfig, *, client: Any = None, registry: WorkflowRegistry | None = None,
                 store: JobStore | None = None, presets: dict | None = None,
                 sleep: Callable[[float], Awaitable[None]] = asyncio.sleep,
                 clock: Callable[[], float] = time.monotonic):
        self.config = config
        self.client = client if client is not None else ComfyClient(config.comfyui_url)
        self.registry = registry if registry is not None else WorkflowRegistry.load()
        self.store = store if store is not None else JobStore(config.output_dir)
        self.presets = presets if presets is not None else load_presets()
        self._sleep = sleep
        self._clock = clock
        self._object_info: dict[str, Any] | None = None
        self._object_info_at = 0.0

    # ------------------------------------------------------------------ status

    async def status(self) -> dict[str, Any]:
        comfy: dict[str, Any] = {"url": self.config.comfyui_url, "reachable": False}
        object_info = None
        try:
            stats = await self.client.system_stats()
            queue = await self.client.queue()
            object_info = await self._cached_object_info()
        except (ComfyUnavailable, ComfyRequestError) as exc:
            comfy["error"] = str(exc)
        else:
            comfy.update(
                reachable=True,
                version=(stats.get("system") or {}).get("comfyui_version"),
                devices=[{"name": device.get("name"),
                          "vram_total_gb": round((device.get("vram_total") or 0) / 2**30, 1),
                          "vram_free_gb": round((device.get("vram_free") or 0) / 2**30, 1)}
                         for device in stats.get("devices") or []],
                queue_running=len(queue.running),
                queue_pending=len(queue.pending),
            )
        workflows = []
        for pack in sorted(self.registry.packs.values(), key=lambda p: (p.capability.name, p.id)):
            info = pack.describe()
            if object_info is not None:
                missing = sorted(pack.required_classes() - set(object_info))
                info.update(available=not missing, missing_node_classes=missing)
            workflows.append(info)
        return {
            "comfyui": comfy,
            "workflows": workflows,
            "rejected_workflows": dict(self.registry.errors),
            "purposes": {name: preset.description for name, preset in self.presets.items()},
            "output_dir": str(self.config.output_dir),
            "export_dir": str(self.config.export_dir),
        }

    async def _cached_object_info(self) -> dict[str, Any]:
        if self._object_info is None or self._clock() - self._object_info_at > OBJECT_INFO_TTL:
            self._object_info = await self.client.object_info()
            self._object_info_at = self._clock()
        return self._object_info

    # -------------------------------------------------------------- generation

    async def generate_image(self, *, prompt: str, asset_name: str, purpose: str = "reference",
                             negative_prompt: str | None = None, width: int | None = None,
                             height: int | None = None, seed: int | None = None, count: int | None = None,
                             workflow: str | None = None, params: dict[str, Any] | None = None,
                             wait_seconds: float = 0) -> dict[str, Any]:
        asset_name = validate_asset_name(asset_name)
        preset = self.presets.get(purpose)
        if preset is None:
            raise ParamError(f"purpose must be one of {sorted(self.presets)}, got {purpose!r}")
        pack = self.registry.get(TEXT_TO_IMAGE.name, workflow)
        composed = compose_prompt(preset, prompt, negative_prompt)
        job_id = new_job_id()
        explicit = {"prompt": composed.positive, "width": width, "height": height, "seed": seed, "count": count}
        if negative_prompt and negative_prompt.strip():
            explicit["negative_prompt"] = composed.negative
        soft = {"seed": random_seed(), "width": preset.width, "height": preset.height,
                "filename_prefix": self._prefix(asset_name, pack, job_id)}
        if composed.negative and "negative_prompt" not in explicit:
            soft["negative_prompt"] = composed.negative
        applied, skipped = self._plan_params(pack, explicit, soft)
        record = {
            "capability": pack.capability.name,
            "purpose": purpose,
            "prompt": {"user": prompt, "positive": composed.positive, "negative": applied.get("negative_prompt")},
        }
        return await self._submit(pack, job_id, asset_name, applied, skipped, params, record, wait_seconds)

    async def generate_model(self, *, asset_name: str | None = None, image_path: str | None = None,
                             image_job_id: str | None = None, image_index: int = 0, seed: int | None = None,
                             target_faces: int | None = None, texture_size: int | None = None,
                             remove_background: bool | None = None, workflow: str | None = None,
                             params: dict[str, Any] | None = None, wait_seconds: float = 0) -> dict[str, Any]:
        if bool(image_path) == bool(image_job_id):
            raise ParamError("pass exactly one of image_path or image_job_id")
        source: dict[str, Any] = {}
        if image_job_id:
            image_job = self.store.load(image_job_id)
            images = (image_job.get("outputs") or {}).get("images") or []
            if image_job.get("status") != "succeeded" or not images:
                raise ParamError(f"image job {image_job_id} is {image_job.get('status')}; wait until it has succeeded")
            if isinstance(image_index, bool) or not isinstance(image_index, int) or not 0 <= image_index < len(images):
                raise ParamError(f"image_index must be between 0 and {len(images) - 1}")
            path = Path(images[image_index])
            asset_name = asset_name or image_job.get("asset_name")
            source.update(image_job_id=image_job["job_id"], image_index=image_index)
        else:
            path = Path(image_path).expanduser()
        if path.suffix.lower() not in IMAGE_SUFFIXES:
            raise ParamError(f"image must be one of {list(IMAGE_SUFFIXES)}, got {path.name}")
        if not path.is_file():
            raise ParamError(f"image not found: {path}")
        asset_name = validate_asset_name(asset_name)
        pack = self.registry.get(IMAGE_TO_3D.name, workflow)
        job_id = new_job_id()
        explicit = {"seed": seed, "target_faces": target_faces, "texture_size": texture_size,
                    "remove_background": remove_background}
        soft = {"seed": random_seed(), "filename_prefix": self._prefix(asset_name, pack, job_id)}
        applied, skipped = self._plan_params(pack, explicit, soft)
        pack.convert_extras(params)  # fail on bad extras before uploading anything
        uploaded = await self.client.upload_image(path)
        applied["image"] = IMAGE_TO_3D.inputs["image"].validate(uploaded)
        copy_name = f"input{path.suffix.lower()}"
        source.update(image_path=str(path.resolve()), uploaded_as=uploaded, copied_as=copy_name)
        record = {"capability": pack.capability.name, "source": source}
        return await self._submit(pack, job_id, asset_name, applied, skipped, params, record, wait_seconds,
                                  copy_inputs={copy_name: path})

    @staticmethod
    def _prefix(asset_name: str, pack: WorkflowPack, job_id: str) -> str:
        return f"metaori/{asset_name}/{pack.capability.kind}_{job_id[:8]}"

    @staticmethod
    def _plan_params(pack: WorkflowPack, explicit: dict[str, Any],
                     soft: dict[str, Any]) -> tuple[dict[str, Any], list[str]]:
        """Explicit params must be supported by the pack; unsupported soft defaults are skipped."""
        explicit = pack.capability.validate(explicit)
        soft = pack.capability.validate(soft)
        unsupported = sorted(key for key in explicit if not pack.supports(key))
        if unsupported:
            raise WorkflowError(
                f"workflow {pack.id} does not support {unsupported}; its inputs are {sorted(pack.inputs)}")
        applied = dict(explicit)
        skipped = []
        for key, value in soft.items():
            if key in applied:
                continue
            if pack.supports(key):
                applied[key] = value
            else:
                skipped.append(key)
        return applied, sorted(skipped)

    async def _submit(self, pack: WorkflowPack, job_id: str, asset_name: str, applied: dict[str, Any],
                      skipped: list[str], extras: dict[str, Any] | None, record: dict[str, Any],
                      wait_seconds: float, copy_inputs: dict[str, Path] | None = None) -> dict[str, Any]:
        graph = pack.bind(applied, extras)
        folder, record = self.store.create(job_id=job_id, kind=pack.capability.kind, asset_name=asset_name, record={
            **record,
            "workflow": {"id": pack.id, "sha256": pack.graph_sha256},
            "params": applied,
            "skipped_params": skipped,
            "extras": pack.resolve_extras(extras),
            "comfyui": {"url": self.config.comfyui_url, "prompt_id": job_id},
            "status": "submitting",
            "queue_position": None,
            "error": None,
            "outputs": {},
        })
        for name, source in (copy_inputs or {}).items():
            shutil.copy2(source, folder / name)
        try:
            await self.client.queue_prompt(graph, job_id)
        except ComfyRequestError as exc:
            return self._finish(record, "failed", error={"message": str(exc), **exc.details})
        except ComfyUnavailable as exc:
            return self._finish(record, "failed", error={"message": str(exc)})
        record["status"] = "queued"
        self.store.save(record)
        return await self.get_job(job_id, wait_seconds=wait_seconds)

    # -------------------------------------------------------------------- jobs

    async def get_job(self, job_id: str, wait_seconds: float = 0) -> dict[str, Any]:
        record = self.store.load(job_id)
        wait = min(max(float(wait_seconds or 0), 0.0), MAX_WAIT_SECONDS)
        deadline = self._clock() + wait
        while record["status"] not in TERMINAL:
            record = await self._refresh(record)
            remaining = deadline - self._clock()
            if record["status"] in TERMINAL or remaining <= 0:
                break
            await self._sleep(min(POLL_INTERVAL, remaining))
        return record

    async def cancel_job(self, job_id: str) -> dict[str, Any]:
        record = self.store.load(job_id)
        if record["status"] in TERMINAL:
            return record
        queue = await self.client.queue()
        if job_id in queue.pending:
            await self.client.delete_pending(job_id)
            return self._finish(record, "cancelled")
        if job_id in queue.running:
            await self.client.interrupt(job_id)
            return await self.get_job(job_id, wait_seconds=15)
        return await self.get_job(job_id)

    async def _refresh(self, record: dict[str, Any]) -> dict[str, Any]:
        job_id = record["job_id"]
        entry = await self.client.history(job_id)
        if entry is not None:
            return await self._complete(record, entry)
        queue = await self.client.queue()
        if job_id in queue.running:
            status, position = "running", None
        elif job_id in queue.pending:
            status, position = "queued", queue.position(job_id)
        else:
            return self._finish(record, "failed", error={
                "reason": "lost",
                "message": "ComfyUI no longer knows this job; it was restarted or its history was cleared"})
        if status == "running" and not record.get("started_at"):
            record["started_at"] = now_iso()
        if record["status"] != status or record.get("queue_position") != position:
            record.update(status=status, queue_position=position)
            self.store.save(record)
        return record

    async def _complete(self, record: dict[str, Any], entry: dict[str, Any]) -> dict[str, Any]:
        status = entry.get("status") or {}
        events = {message[0]: message[1] for message in status.get("messages") or []
                  if isinstance(message, (list, tuple)) and len(message) == 2 and isinstance(message[1], dict)}
        timing = _execution_seconds(events)
        if "execution_interrupted" in events:
            return self._finish(record, "cancelled", execution_s=timing)
        if status.get("status_str") != "success":
            error = events.get("execution_error") or {}
            return self._finish(record, "failed", execution_s=timing, error={
                "message": error.get("exception_message") or "ComfyUI reported an error without details",
                "node_id": error.get("node_id"),
                "node_type": error.get("node_type"),
                "exception_type": error.get("exception_type"),
            })
        pack = self.registry.packs.get(record["workflow"]["id"])
        if pack is None:
            return self._finish(record, "failed", execution_s=timing, error={
                "message": f"workflow {record['workflow']['id']} is no longer installed; its outputs cannot be mapped"})
        folder = Path(record["folder"])
        node_outputs = entry.get("outputs") or {}
        outputs: dict[str, Any] = {}
        for name, spec in pack.outputs.items():
            files = (node_outputs.get(spec.node) or {}).get(spec.key) or []
            if not files:
                return self._finish(record, "failed", execution_s=timing, outputs=outputs, error={
                    "message": f"ComfyUI finished but node {spec.node} ({spec.class_type}) returned no {spec.key!r} output"})
            paths = []
            for index, item in enumerate(files, start=1):
                dest = folder / _output_name(name, index, len(files), item["filename"])
                try:
                    await self.client.download(item["filename"], item.get("subfolder", ""), item.get("type", "output"), dest)
                except (ComfyRequestError, OSError) as exc:
                    return self._finish(record, "failed", execution_s=timing, outputs=outputs,
                                        error={"message": f"downloading {item['filename']} failed: {exc}"})
                paths.append(str(dest))
            outputs[name] = paths if name == "images" else paths[0]
        return self._finish(record, "succeeded", execution_s=timing, outputs=outputs)

    def _finish(self, record: dict[str, Any], status: str, *, error: dict[str, Any] | None = None,
                outputs: dict[str, Any] | None = None, execution_s: float | None = None) -> dict[str, Any]:
        record["status"] = status
        record["queue_position"] = None
        if error is not None:
            record["error"] = error
        if outputs is not None:
            record["outputs"] = outputs
        if execution_s is not None:
            record["execution_s"] = execution_s
        record["finished_at"] = now_iso()
        record["duration_s"] = _seconds_between(record.get("created_at"), record["finished_at"])
        self.store.save(record)
        return record
```

```python
"""Run MetaOri's bpy scripts inside Blender through the addon's execute_code command.

Each script in bpy_scripts/ defines run(params) -> dict. build_code() appends a runner that
calls it with the JSON parameters and prints one marker line holding the JSON result or the
error. The runner never lets an exception escape execute_code, because an addon-side error
makes the MCP server drop its socket to Blender.
"""
from __future__ import annotations

import json
from pathlib import Path
from typing import Any

from .errors import BlenderScriptError

SCRIPTS_DIR = Path(__file__).with_name("bpy_scripts")
RESULT_MARK = "METAORI_RESULT:"
ERROR_MARK = "METAORI_ERROR:"

_RUNNER = """


def _metaori_main():
    import json as _json
    import traceback as _traceback
    try:
        _line = {result_mark} + _json.dumps(run(_json.loads({params})))
    except Exception as _exc:
        _line = {error_mark} + _json.dumps({{"type": type(_exc).__name__, "message": str(_exc),
                                             "traceback": _traceback.format_exc()}})
    print(_line)


_metaori_main()
"""


def build_code(script: str, params: dict[str, Any], scripts_dir: Path = SCRIPTS_DIR) -> str:
    source = (Path(scripts_dir) / f"{script}.py").read_text(encoding="utf-8")
    return source + _RUNNER.format(params=repr(json.dumps(params)), result_mark=repr(RESULT_MARK),
                                   error_mark=repr(ERROR_MARK))


def parse_output(script: str, output: str) -> dict[str, Any]:
    for line in reversed(output.splitlines()):
        if line.startswith(RESULT_MARK):
            return json.loads(line[len(RESULT_MARK):])
        if line.startswith(ERROR_MARK):
            error = json.loads(line[len(ERROR_MARK):])
            raise BlenderScriptError(
                f"{script} failed in Blender: {error.get('type')}: {error.get('message')}\n"
                f"{error.get('traceback', '')}".rstrip())
    raise BlenderScriptError(f"{script} printed no result; Blender output:\n{output[-2000:]}")


def run_script(blender: Any, script: str, params: dict[str, Any], scripts_dir: Path = SCRIPTS_DIR) -> dict[str, Any]:
    """Run bpy_scripts/<script>.py in the connected Blender and return its result dict."""
    code = build_code(script, params, scripts_dir)
    try:
        response = blender.send_command("execute_code", {"code": code})
    except Exception as exc:
        raise BlenderScriptError(f"Blender could not run {script}: {exc}") from exc
    output = response.get("result", "") if isinstance(response, dict) else str(response)
    return parse_output(script, output)
```

```python
"""import_model - MetaOri bpy script: import a model file into the open Blender scene.

Runs inside Blender. MetaOri sends this file plus a runner (blender_ops.build_code) through the
addon's execute_code command, and run(params) returns a JSON-serialisable dict.
Offline: blender -b scene.blend -P import_model.py -- --params params.json

params: file_path (absolute .glb/.gltf/.fbx/.obj), name (optional new name for a single root).
"""
import json
import os
import sys

import bpy
import numpy as np
from mathutils import Vector


def _supported(operator, wanted):
    """Keep only the keyword arguments this Blender version's operator accepts."""
    names = set(operator.get_rna_type().properties.keys())
    return {key: value for key, value in wanted.items() if key in names}


def _import(path):
    extension = os.path.splitext(path)[1].lower()
    if extension in (".glb", ".gltf"):
        operator = bpy.ops.import_scene.gltf
        # merge_vertices re-joins the vertices glTF splits along UV seams.
        operator(**_supported(operator, {"filepath": path, "merge_vertices": True,
                                         "import_scene_as_collection": False}))
    elif extension == ".fbx":
        bpy.ops.import_scene.fbx(filepath=path)
    elif extension == ".obj":
        bpy.ops.wm.obj_import(filepath=path)
    else:
        raise ValueError(f"unsupported model format {extension!r}; use .glb, .gltf, .fbx or .obj")


def _tri_count(mesh):
    totals = np.empty(len(mesh.polygons), dtype=np.int32)
    mesh.polygons.foreach_get("loop_total", totals)
    return int((totals - 2).sum())


def _world_bbox(objects):
    corners = [obj.matrix_world @ Vector(corner) for obj in objects for corner in obj.bound_box]
    if not corners:
        return None
    low = [min(corner[i] for corner in corners) for i in range(3)]
    high = [max(corner[i] for corner in corners) for i in range(3)]
    return {"min": [round(v, 5) for v in low], "max": [round(v, 5) for v in high],
            "dimensions": [round(high[i] - low[i], 5) for i in range(3)]}


def _images(materials):
    found = {}
    for material in materials:
        if material.node_tree is None:
            continue
        for node in material.node_tree.nodes:
            if node.type == "TEX_IMAGE" and node.image is not None:
                found[node.image.name] = list(node.image.size)
    return [{"name": name, "size": size} for name, size in sorted(found.items())]

    path = params["file_path"]
    if not os.path.isfile(path):
        raise FileNotFoundError(path)
    before = set(bpy.data.objects)
    _import(path)
    bpy.context.view_layer.update()
    created = [obj for obj in bpy.data.objects if obj not in before]
    if not created:
        raise RuntimeError(f"the importer created no objects from {path}")
    roots = [obj for obj in created if obj.parent is None or obj.parent not in created]
    if params.get("name") and len(roots) == 1:
        roots[0].name = params["name"]
    meshes = [obj for obj in created if obj.type == "MESH"]
    materials = {slot.material for obj in meshes for slot in obj.material_slots if slot.material is not None}
    return {
        "file_path": path,
        "objects": [{"name": obj.name, "type": obj.type, "parent": obj.parent.name if obj.parent else None}
                    for obj in created],
        "roots": [obj.name for obj in roots],
        "mesh_objects": [obj.name for obj in meshes],
        "tris": sum(_tri_count(obj.data) for obj in meshes),
        "world_bbox": _world_bbox(meshes),
        "materials": sorted(material.name for material in materials),
        "images": _images(materials),
    }
```

```python
"""normalize - MetaOri bpy script: bring a model to the asset standard and export it for UE.

Runs inside Blender. MetaOri sends this file plus a runner (blender_ops.build_code) through the
addon's execute_code command, and run(params) returns a JSON-serialisable report.
Offline: blender -b scene.blend -P normalize.py -- --params params.json

Order: collect meshes -> apply modifiers -> unparent -> join -> rotate -> bake transforms ->
scale to size -> merge doubles / delete loose -> drop small islands -> recalc normals ->
decimate -> smooth -> pivot to the bottom centre at the origin -> rename -> export FBX.
"""
import json
import math
import os
import sys

import bmesh
import bpy
import numpy as np
from mathutils import Euler, Matrix, Vector

# The UE_VR_Game preset (Tools/blender/export_fbx.py): the scene is in metres and UE reads raw FBX
# values as centimetres, so FBX_SCALE_NONE with global_scale 1.0 writes centimetres.
UE_FBX_SETTINGS = dict(object_types={"MESH"}, global_scale=1.0, apply_unit_scale=True,
                       apply_scale_options="FBX_SCALE_NONE", axis_forward="-Z", axis_up="Y",
                       mesh_smooth_type="FACE", use_mesh_modifiers=True, add_leaf_bones=False,
                       bake_anim=False)
SIZE_AXES = {"width": 0, "depth": 1, "height": 2}
DEFAULTS = dict(apply_modifiers=True, rotate_deg=None, target_size_m=None, size_axis="height",
                merge_distance_m=0.0001, min_island_ratio=0.01, recalc_normals=True, target_tris=None,
                smooth_angle_deg=30.0, rename=True, export_path=None, overwrite_export=False)


def _round(values, digits=5):
    return [round(float(value), digits) for value in values]


def _coords(obj):
    """World-space vertex positions as an (n, 3) array."""
    mesh = obj.data
    flat = np.empty(len(mesh.vertices) * 3, dtype=np.float32)
    mesh.vertices.foreach_get("co", flat)
    local = flat.reshape(-1, 3).astype(np.float64)
    matrix = np.array(obj.matrix_world, dtype=np.float64)
    return local @ matrix[:3, :3].T + matrix[:3, 3]


def _tri_count(mesh):
    totals = np.empty(len(mesh.polygons), dtype=np.int32)
    mesh.polygons.foreach_get("loop_total", totals)
    return int((totals - 2).sum())


def _face_islands(bm):
    """Groups of faces connected through shared edges."""
    bm.faces.index_update()
    seen = bytearray(len(bm.faces))
    islands = []
    for face in bm.faces:
        if seen[face.index]:
            continue
        seen[face.index] = 1
        stack, island = [face], []
        while stack:
            current = stack.pop()
            island.append(current)
            for edge in current.edges:
                for neighbour in edge.link_faces:
                    if not seen[neighbour.index]:
                        seen[neighbour.index] = 1
                        stack.append(neighbour)
        islands.append(island)
    return islands


def _stats(objects, with_islands=False):
    points = [_coords(obj) for obj in objects if len(obj.data.vertices)]
    if points:
        stacked = np.concatenate(points)
        low, high = stacked.min(axis=0), stacked.max(axis=0)
    else:
        low = high = np.zeros(3)
    stats = {
        "dimensions_m": _round(high - low), "bbox_min": _round(low), "bbox_max": _round(high),
        "verts": 0, "faces": 0, "tris": 0, "non_manifold_edges": 0,
        "uv_layers": sorted({layer.name for obj in objects for layer in obj.data.uv_layers}),
        "materials": sorted({slot.material.name for obj in objects for slot in obj.material_slots if slot.material}),
        "custom_normals": any(getattr(obj.data, "has_custom_normals", False) for obj in objects),
    }
    if with_islands:
        stats["islands"] = 0
    for obj in objects:
        mesh = obj.data
        stats["verts"] += len(mesh.vertices)
        stats["faces"] += len(mesh.polygons)
        stats["tris"] += _tri_count(mesh)
        bm = bmesh.new()
        try:
            bm.from_mesh(mesh)
            stats["non_manifold_edges"] += sum(1 for edge in bm.edges if not edge.is_manifold)
            if with_islands:
                stats["islands"] += len(_face_islands(bm))
        finally:
            bm.free()
    if len(objects) == 1:
        stats["location"] = _round(objects[0].location)
    return stats


def _collect_meshes(obj):
    meshes = [obj] if obj.type == "MESH" else []
    return meshes + [child for child in obj.children_recursive if child.type == "MESH"]


def _bake_modifiers(obj, depsgraph):
    """Replace the mesh with its evaluated version (all modifiers applied)."""
    evaluated = obj.evaluated_get(depsgraph)
    mesh = bpy.data.meshes.new_from_object(evaluated, preserve_all_data_layers=True, depsgraph=depsgraph)
    old = obj.data
    obj.modifiers.clear()
    obj.data = mesh
    if old.users == 0:
        bpy.data.meshes.remove(old)


def _clear_custom_normals(obj):
    mesh = obj.data
    if not getattr(mesh, "has_custom_normals", False):
        return
    attribute = mesh.attributes.get("custom_normal")  # Blender 5 stores custom normals as an attribute
    if attribute is not None:
        mesh.attributes.remove(attribute)
    if mesh.has_custom_normals:
        with bpy.context.temp_override(object=obj, active_object=obj, selected_objects=[obj],
                                       selected_editable_objects=[obj]):
            bpy.ops.mesh.customdata_custom_splitnormals_clear()


def _smooth(obj, angle_deg):
    mesh = obj.data
    _clear_custom_normals(obj)
    if hasattr(mesh, "set_sharp_from_angle"):  # Blender 4.1+
        mesh.shade_smooth()
        mesh.set_sharp_from_angle(angle=math.radians(angle_deg))
    else:
        for polygon in mesh.polygons:
            polygon.use_smooth = True
        mesh.use_auto_smooth = True
        mesh.auto_smooth_angle = math.radians(angle_deg)


def _used_elsewhere(material, mesh):
    return any(other is not mesh and other.users > 0 and material in other.materials[:] for other in bpy.data.meshes)


def _export(obj, path):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    view_layer = bpy.context.view_layer
    previous = [other for other in view_layer.objects if other.select_get()]
    active = view_layer.objects.active
    for other in previous:
        other.select_set(False)
    obj.select_set(True)
    view_layer.objects.active = obj
    try:
        bpy.ops.export_scene.fbx(filepath=path, use_selection=True, **UE_FBX_SETTINGS)
    finally:
        obj.select_set(False)
        for other in previous:
            other.select_set(True)
        view_layer.objects.active = active
    return {"path": path, "bytes": os.path.getsize(path), "preset": "ue"}


def run(params):
    p = dict(DEFAULTS)
    p.update(params)
    obj = bpy.data.objects.get(p["object_name"])
    if obj is None:
        raise ValueError(f"object {p['object_name']!r} not found")
    asset = p["asset_name"]
    target_name = f"SM_{asset}"
    steps, warnings = [], []
    scale_length = bpy.context.scene.unit_settings.scale_length
    if abs(scale_length - 1.0) > 1e-6:
        warnings.append(f"scene unit scale is {scale_length}, not 1.0; sizes are in Blender units")

    meshes = _collect_meshes(obj)
    if not meshes:
        raise ValueError(f"{obj.name} has no mesh objects")
    for mesh_obj in meshes:
        if mesh_obj.find_armature() is not None or any(m.type == "ARMATURE" for m in mesh_obj.modifiers):
            raise ValueError(f"{mesh_obj.name} is skinned; normalization is for static meshes")
    if p["target_tris"] and not p["apply_modifiers"] and any(mesh_obj.modifiers for mesh_obj in meshes):
        raise ValueError("target_tris needs apply_modifiers=true when the mesh has modifiers")
    if p["rename"]:
        clash = bpy.data.objects.get(target_name)
        if clash is not None and clash not in meshes:
            raise ValueError(f"an object named {target_name} already exists; rename or delete it first")
    export_path = p["export_path"]
    if export_path and os.path.exists(export_path) and not p["overwrite_export"]:
        raise FileExistsError(f"{export_path} exists; pass overwrite_export=true to replace it")

    before = _stats(meshes)

    for mesh_obj in meshes:
        if mesh_obj.data.users > 1:
            mesh_obj.data = mesh_obj.data.copy()
    if p["apply_modifiers"]:
        with_modifiers = [mesh_obj for mesh_obj in meshes if mesh_obj.modifiers]
        depsgraph = bpy.context.evaluated_depsgraph_get()
        for mesh_obj in with_modifiers:
            _bake_modifiers(mesh_obj, depsgraph)
        if with_modifiers:
            steps.append({"step": "apply_modifiers", "objects": len(with_modifiers)})

    for mesh_obj in meshes:
        if mesh_obj.parent is not None:
            world = mesh_obj.matrix_world.copy()
            mesh_obj.parent = None
            mesh_obj.matrix_world = world

    target = obj if obj.type == "MESH" else meshes[0]
    if len(meshes) > 1:
        with bpy.context.temp_override(active_object=target, object=target, selected_objects=meshes,
                                       selected_editable_objects=meshes):
            bpy.ops.object.join()
        steps.append({"step": "join", "objects": len(meshes)})
    if obj is not target:
        warnings.append(f"root {obj.type.lower()} {obj.name!r} was left in place; its meshes were joined")

    mesh = target.data
    matrix = target.matrix_world.copy()
    if p["rotate_deg"]:
        if len(p["rotate_deg"]) != 3:
            raise ValueError("rotate_deg needs three angles [x, y, z]")
        rotation = Euler([math.radians(angle) for angle in p["rotate_deg"]], "XYZ").to_matrix().to_4x4()
        matrix = rotation @ matrix
        steps.append({"step": "rotate", "deg": _round(p["rotate_deg"], 3)})
    mesh.transform(matrix)
    if matrix.determinant() < 0 and hasattr(mesh, "flip_normals"):
        mesh.flip_normals()
    target.matrix_world = Matrix.Identity(4)
    steps.append({"step": "bake_transform"})

    if p["target_size_m"] is not None:
        size = float(p["target_size_m"])
        axis = p["size_axis"]
        if size <= 0:
            raise ValueError("target_size_m must be positive")
        if axis != "longest" and axis not in SIZE_AXES:
            raise ValueError(f"size_axis must be height, width, depth or longest, got {axis!r}")
        coords = _coords(target)
        dims = coords.max(axis=0) - coords.min(axis=0)
        current = float(dims.max() if axis == "longest" else dims[SIZE_AXES[axis]])
        if current <= 0:
            raise ValueError(f"the model has no extent along {axis}")
        factor = size / current
        mesh.transform(Matrix.Scale(factor, 4))
        steps.append({"step": "scale", "axis": axis, "target_m": size, "factor": round(factor, 6)})

    bm = bmesh.new()
    try:
        bm.from_mesh(mesh)
        verts_before = len(bm.verts)
        if p["merge_distance_m"] and p["merge_distance_m"] > 0:
            bmesh.ops.remove_doubles(bm, verts=list(bm.verts), dist=float(p["merge_distance_m"]))
        merged = verts_before - len(bm.verts)
        loose_edges = [edge for edge in bm.edges if not edge.link_faces]
        if loose_edges:
            bmesh.ops.delete(bm, geom=loose_edges, context="EDGES")
        loose_verts = [vert for vert in bm.verts if not vert.link_edges]
        if loose_verts:
            bmesh.ops.delete(bm, geom=loose_verts, context="VERTS")
        steps.append({"step": "merge_and_loose", "merged_verts": merged, "loose_edges": len(loose_edges),
                      "loose_verts": len(loose_verts)})
        ratio = float(p["min_island_ratio"] or 0)
        if ratio > 0 and len(bm.faces):
            islands = _face_islands(bm)
            largest = max(len(island) for island in islands)
            small = [island for island in islands if len(island) < ratio * largest]
            doomed = [face for island in small for face in island]
            if doomed:
                bmesh.ops.delete(bm, geom=doomed, context="FACES")
            steps.append({"step": "islands", "found": len(islands), "removed": len(small),
                          "removed_faces": len(doomed)})
        if p["recalc_normals"]:
            bmesh.ops.recalc_face_normals(bm, faces=list(bm.faces))
            steps.append({"step": "recalc_normals"})
        bm.to_mesh(mesh)
    finally:
        bm.free()
    mesh.update()

    if p["target_tris"]:
        budget = int(p["target_tris"])
        tris = _tri_count(mesh)
        if tris > budget:
            modifier = target.modifiers.new("MetaOriDecimate", "DECIMATE")
            modifier.decimate_type = "COLLAPSE"
            modifier.ratio = budget / tris
            modifier.use_collapse_triangulate = True
            _bake_modifiers(target, bpy.context.evaluated_depsgraph_get())
            mesh = target.data
            steps.append({"step": "decimate", "ratio": round(budget / tris, 5), "tris_before": tris,
                          "tris_after": _tri_count(mesh)})
        else:
            steps.append({"step": "decimate", "skipped": f"{tris} tris already within {budget}"})

    if p["smooth_angle_deg"] is not None:
        _smooth(target, float(p["smooth_angle_deg"]))
        steps.append({"step": "smooth", "angle_deg": float(p["smooth_angle_deg"])})

    coords = _coords(target)
    if not len(coords):
        raise ValueError("no geometry is left after cleanup")
    low, high = coords.min(axis=0), coords.max(axis=0)
    pivot = Vector(((low[0] + high[0]) / 2, (low[1] + high[1]) / 2, low[2]))
    mesh.transform(Matrix.Translation(-pivot))
    mesh.update()
    target.matrix_world = Matrix.Identity(4)
    steps.append({"step": "pivot", "moved_by": _round(-pivot)})

    if p["rename"]:
        target.name = target_name
        mesh.name = target_name
        materials = []
        for slot in target.material_slots:
            if slot.material is not None and slot.material not in materials:
                materials.append(slot.material)
        renamed = []
        for index, material in enumerate(materials, start=1):
            if _used_elsewhere(material, mesh):
                copy = material.copy()
                for slot in target.material_slots:
                    if slot.material == material:
                        slot.material = copy
                material = copy
            material.name = f"MI_{asset}" if len(materials) == 1 else f"MI_{asset}_{index:02d}"
            renamed.append(material.name)
        steps.append({"step": "rename", "object": target.name, "materials": renamed})
        if target.name != target_name:
            warnings.append(f"the object is named {target.name!r} because {target_name} is taken")

    export = None
    if export_path:
        export = _export(target, export_path)
        steps.append({"step": "export", **export})

    return {"object": target.name, "mesh": target.data.name, "asset_name": asset, "before": before,
            "after": _stats([target], with_islands=True), "steps": steps, "export": export,
            "warnings": warnings}


if __name__ == "__main__" and "--params" in sys.argv:
    with open(sys.argv[sys.argv.index("--params") + 1], encoding="utf-8") as _handle:
        print("METAORI_RESULT:" + json.dumps(run(json.load(_handle))))
```

```python
import json
from pathlib import Path
from typing import Any, Callable

from .blender_ops import run_script
from .capabilities import validate_asset_name
from .config import MetaOriConfig
from .errors import ParamError
from .service import MetaOriService

MODEL_SUFFIXES = (".glb", ".gltf", ".fbx", ".obj")
SIZE_AXES = ("height", "width", "depth", "longest")


def _json(data: Any) -> str:
    return json.dumps(data, ensure_ascii=False, indent=2)


def register_tools(mcp: Any, get_blender_connection: Callable[[], Any],
                   service: MetaOriService | None = None) -> dict[str, Callable]:
    """Register the MetaOri tools on a FastMCP server and return them by name."""
    holder = {"service": service}

    def svc() -> MetaOriService:
        if holder["service"] is None:  # built on first use, so importing the server reads no env
            holder["service"] = MetaOriService(MetaOriConfig.from_env())
        return holder["service"]

    @mcp.tool()
    async def metaori_status() -> str:
        """MetaOri 3D - local AI asset generation on ComfyUI. Call this first.

        Reports whether ComfyUI is reachable (version, GPU memory, queue), each workflow with its
        standard inputs, extra `params` (defaults, allowed values) and missing node classes, the
        prompt purposes, and the folders for job outputs and FBX exports.

        Pipeline: metaori_generate_image (purpose="reference") -> metaori_get_job until succeeded ->
        metaori_generate_model(image_job_id=...) -> metaori_get_job -> metaori_import_model ->
        get_viewport_screenshot -> metaori_normalize_model. Generate one object at a time.
        """
        return _json(await svc().status())

    @mcp.tool()
    async def metaori_generate_image(prompt: str, asset_name: str, purpose: str = "reference",
                                     negative_prompt: str | None = None, width: int | None = None,
                                     height: int | None = None, seed: int | None = None,
                                     count: int | None = None, workflow: str | None = None,
                                     params: dict[str, Any] | None = None, wait_seconds: float = 0) -> str:
        """MetaOri function 1: text-to-image on the local ComfyUI (reference pictures or textures).

        - prompt: what to draw, Chinese or English; describe one object (shape, material, style).
        - asset_name: PascalCase asset name such as BronzeCenser or WoodCrate_A. Outputs go to
          <output_dir>/<asset_name>/<time>_t2i_<job>/ with job.json (prompt, seed, params, workflow).
        - purpose: "reference" (default) adds a single-object, white-background, 3/4-view template so
          the picture works for metaori_generate_model; "texture" asks for a flat, evenly lit,
          top-down surface (seamless tiling is not guaranteed); "raw" sends the prompt unchanged.
        - negative_prompt: extra things to avoid, added to the purpose's negative prompt.
        - width, height: multiples of 8 in 256-2048 (default 1024x1024). count: 1-4 images.
        - seed: omit for a random one (recorded). workflow: a workflow id from metaori_status.
          params: that workflow's extras, e.g. {"steps": 30}.
        - wait_seconds: wait up to this long (max 600) for the result; 0 returns at once.

        Returns the job record; outputs.images lists the PNG paths once status is "succeeded".
        A warm run takes about 15 s; the first run after ComfyUI starts loads models (minutes).
        """
        return _json(await svc().generate_image(
            prompt=prompt, asset_name=asset_name, purpose=purpose, negative_prompt=negative_prompt,
            width=width, height=height, seed=seed, count=count, workflow=workflow, params=params,
            wait_seconds=wait_seconds))

    @mcp.tool()
    async def metaori_generate_model(asset_name: str | None = None, image_path: str | None = None,
                                     image_job_id: str | None = None, image_index: int = 0,
                                     seed: int | None = None, target_faces: int | None = None,
                                     texture_size: int | None = None, remove_background: bool | None = None,
                                     workflow: str | None = None, params: dict[str, Any] | None = None,
                                     wait_seconds: float = 0) -> str:
        """MetaOri function 2: image-to-3D on the local ComfyUI; produces one textured GLB.

        - image_job_id: a succeeded metaori_generate_image job (image_index picks the picture and
          asset_name defaults to that job's), OR image_path: a local .png/.jpg/.jpeg/.webp file.
        - asset_name: PascalCase asset name; required with image_path.
        - target_faces: face count after the workflow's decimation (default workflow: 700000). For
          props 30000-100000 bakes cleaner textures than decimating later in Blender.
        - texture_size: 512, 1024, 2048 or 4096 (default 4096; B-tier assets use 2048 at most).
        - remove_background: true by default in the Pixal3D workflow.
        - seed: omit for a random one. workflow / params: see metaori_status,
          e.g. {"model_variant": "trellis2"}.
        - wait_seconds: wait up to this long (max 600); generation takes minutes, so poll with
          metaori_get_job.

        Returns the job record; outputs.model is the GLB path once status is "succeeded".
        """
        return _json(await svc().generate_model(
            asset_name=asset_name, image_path=image_path, image_job_id=image_job_id, image_index=image_index,
            seed=seed, target_faces=target_faces, texture_size=texture_size, remove_background=remove_background,
            workflow=workflow, params=params, wait_seconds=wait_seconds))

    @mcp.tool()
    async def metaori_get_job(job_id: str, wait_seconds: float = 60) -> str:
        """Check a MetaOri job, waiting up to wait_seconds (max 600) for it to finish.

        status is queued (with queue_position), running, succeeded, failed or cancelled. On success
        `outputs` holds local paths (images: list, model: GLB path); a failure carries `error` with
        the ComfyUI node and message. Finished jobs are read from job.json without asking ComfyUI.
        """
        return _json(await svc().get_job(job_id, wait_seconds=wait_seconds))

    @mcp.tool()
    async def metaori_cancel_job(job_id: str) -> str:
        """Cancel a queued or running MetaOri job; returns the job record."""
        return _json(await svc().cancel_job(job_id))

    @mcp.tool()
    async def metaori_import_model(job_id: str | None = None, file_path: str | None = None,
                                   name: str | None = None) -> str:
        """Import into the open Blender scene the GLB of a succeeded metaori_generate_model job
        (job_id) or any .glb/.gltf/.fbx/.obj file (file_path).

        name renames the imported root object when there is exactly one. Returns the created
        objects, the root and mesh object names, their combined world bounding box in metres, the
        triangle count, materials and texture images. Check the facing with get_viewport_screenshot
        before metaori_normalize_model.
        """
        if bool(job_id) == bool(file_path):
            raise ParamError("pass exactly one of job_id or file_path")
        if job_id:
            record = await svc().get_job(job_id)
            model = (record.get("outputs") or {}).get("model")
            if record["status"] != "succeeded" or not model:
                raise ParamError(f"job {job_id} is {record['status']} and has no model to import")
            path = Path(model)
        else:
            path = Path(file_path).expanduser()
        if path.suffix.lower() not in MODEL_SUFFIXES:
            raise ParamError(f"file must be one of {list(MODEL_SUFFIXES)}, got {path.name}")
        if not path.is_file():
            raise ParamError(f"model file not found: {path}")
        params = {"file_path": str(path.resolve()), "name": name}
        return _json(await asyncio.to_thread(run_script, get_blender_connection(), "import_model", params))

    @mcp.tool()
    async def metaori_normalize_model(object_name: str, asset_name: str, target_size_m: float | None = None,
                                      size_axis: str = "height", rotate_deg: list[float] | None = None,
                                      target_tris: int | None = None, merge_distance_m: float = 0.0001,
                                      min_island_ratio: float = 0.01, recalc_normals: bool = True,
                                      smooth_angle_deg: float | None = 30.0, apply_modifiers: bool = True,
                                      rename: bool = True, export_fbx: bool = True,
                                      overwrite_export: bool = False) -> str:
        """MetaOri function 3: normalize a Blender model to the asset standard and export FBX for UE.

        Works on a mesh object, or an empty whose child meshes are joined into one. In order:
        apply modifiers, unparent and join; rotate by rotate_deg [x, y, z] degrees (turn the
        model's front to -Y); bake transforms; scale uniformly so size_axis ("height" = Z,
        "width" = X, "depth" = Y or "longest") equals target_size_m metres; merge vertices closer
        than merge_distance_m and delete loose geometry; drop islands with fewer faces than
        min_island_ratio x the largest (0 keeps all); recalculate normals outward; decimate to
        target_tris; clear custom normals and smooth by smooth_angle_deg (None keeps the imported
        normals); put the pivot at the bottom centre on the world origin; rename to
        SM_<asset_name> / MI_<asset_name>; export <export_dir>/<asset_name>/SM_<asset_name>.fbx with
        the UE preset (Blender metres become UE centimetres). An existing FBX is only replaced
        with overwrite_export=true.

        Returns before/after stats (size in metres, tris, non-manifold edges, islands), each
        step's details, the export path and warnings.
        """
        asset_name = validate_asset_name(asset_name)
        if size_axis not in SIZE_AXES:
            raise ParamError(f"size_axis must be one of {list(SIZE_AXES)}, got {size_axis!r}")
        if target_size_m is not None and target_size_m <= 0:
            raise ParamError("target_size_m must be positive")
        if rotate_deg is not None and len(rotate_deg) != 3:
            raise ParamError("rotate_deg needs three angles [x, y, z]")
        if target_tris is not None and target_tris < 4:
            raise ParamError("target_tris must be at least 4")
        if merge_distance_m < 0:
            raise ParamError("merge_distance_m must not be negative")
        if not 0 <= min_island_ratio < 1:
            raise ParamError("min_island_ratio must be in [0, 1)")
        export_path = None
        if export_fbx:
            export_path = svc().config.export_dir / asset_name / f"SM_{asset_name}.fbx"
            if export_path.exists() and not overwrite_export:
                raise ParamError(f"{export_path} already exists; pass overwrite_export=true to replace it")
        params = {
            "object_name": object_name, "asset_name": asset_name, "apply_modifiers": apply_modifiers,
            "rotate_deg": rotate_deg, "target_size_m": target_size_m, "size_axis": size_axis,
            "merge_distance_m": merge_distance_m, "min_island_ratio": min_island_ratio,
            "recalc_normals": recalc_normals, "target_tris": target_tris, "smooth_angle_deg": smooth_angle_deg,
            "rename": rename, "export_path": str(export_path) if export_path else None,
            "overwrite_export": overwrite_export,
        }
        return _json(await asyncio.to_thread(run_script, get_blender_connection(), "normalize", params))

    return {tool.__name__: tool for tool in (
        metaori_status, metaori_generate_image, metaori_generate_model, metaori_get_job,
        metaori_cancel_job, metaori_import_model, metaori_normalize_model)}
```

~~~markdown
# MetaOri 3D

MetaOri 3D 是本 fork 在 blender-mcp 服务端加入的模块：以本机 ComfyUI 为生产后台，通过 MCP 提供文生图、图生模型和模型规范化。设计见 `docs/superpowers/specs/2026-09-28-metaori-3d-design.md`。

## 配置

| 环境变量 | 默认 | 说明 |
|---|---|---|
| `METAORI_COMFYUI_URL` | `http://127.0.0.1:8188` | ComfyUI 地址 |
| `METAORI_OUTPUT_DIR` | `~/MetaOri/AI_Raw` | 任务目录根（每个任务一个子目录，永不覆盖） |
| `METAORI_EXPORT_DIR` | `~/MetaOri/Export` | 规范化后 FBX 的导出根目录 |

相对路径按 MCP 服务进程的工作目录解析，建议写绝对路径。以 fork 源码运行的 MCP 客户端配置：

```json
"blender": {
  "command": "uv",
  "args": ["run", "--frozen", "--project", "<fork 路径>", "mcp-for-blender"],
  "env": {"METAORI_OUTPUT_DIR": "<项目>/SourceArt/AI_Raw", "METAORI_EXPORT_DIR": "<项目>/SourceArt/Export"}
}
```

Blender 插件无需改动或重装：MetaOri 只通过插件已有的 `execute_code` 命令运行 `bpy_scripts/` 下的脚本。

## 工具

| 工具 | 作用 |
|---|---|
| `metaori_status` | ComfyUI 状态、工作流与参数、输出目录 |
| `metaori_generate_image` | 功能 1：文生图（`purpose`：reference / texture / raw） |
| `metaori_generate_model` | 功能 2：图生模型（`image_job_id` 或 `image_path`） |
| `metaori_get_job` | 查询或等待任务；完成时输出已下载到任务目录 |
| `metaori_cancel_job` | 取消排队或运行中的任务 |
| `metaori_import_model` | 把任务 GLB 或任意模型文件导入 Blender |
| `metaori_normalize_model` | 功能 3：规范化并按 UE 预设导出 FBX |

典型流程：

1. `metaori_status`：确认 ComfyUI 可达、工作流节点齐全。
2. `metaori_generate_image(prompt=..., asset_name="BronzeCenser")`，再 `metaori_get_job(job_id, wait_seconds=120)`，查看 `outputs.images`。
3. `metaori_generate_model(image_job_id=..., target_faces=80000, texture_size=2048)`，再多次 `metaori_get_job(job_id, wait_seconds=300)`。
4. `metaori_import_model(job_id=...)`，用 `get_viewport_screenshot` 判断朝向。
5. `metaori_normalize_model(object_name=<导入的根对象>, asset_name="BronzeCenser", target_size_m=0.35, rotate_deg=[0, 0, 0], target_tris=50000)`。

## 任务目录

```
<METAORI_OUTPUT_DIR>/<AssetName>/<YYYYMMDD-HHMMSS>_<t2i|i2m>_<job8>/
    image_01.png ...        文生图输出
    input.png, model.glb    图生模型的参考图副本与输出
    job.json
```

`job.json` 记录提示词（原文与最终版本）、实际生效的参数（含 seed）、被跳过的参数、工作流 id 与 sha256、ComfyUI prompt_id、状态、耗时和输出路径。

## 更换或新增工作流

1. 在 ComfyUI 中搭好工作流，用菜单 “导出（API）” 得到 API 格式 JSON。
2. 新建 `src/blender_mcp/metaori/workflows/<id>/`（目录名即 id），放入 `workflow_api.json`。
3. 写 `manifest.json`（参考现有两个包）：
   - `capability`：`text_to_image` 或 `image_to_3d`；
   - `inputs`：标准参数 → 目标列表 `[{"node", "class_type", "input"}]`，一个参数可以绑多个节点；
   - `extras`：工作流特有参数（`type`、`default`、`min`/`max` 或枚举 `values`、`bind`），`default` 必须与图里的当前值一致；
   - `outputs`：标准输出 → `{"node", "class_type", "key"}`（图片保存节点用 `images`，3D 保存节点用 `3d`）；
   - 同一能力只能有一个 `"default": true`。
4. 重启 MCP 服务，用 `metaori_status` 检查：加载失败的包列在 `rejected_workflows` 并写明原因，缺失的节点类列在 `missing_node_classes`。
5. 运行 `uv run --frozen --with pytest pytest tests/test_metaori_workflows.py -q`；若替换了默认工作流，同步更新 `test_shipped_packs_load_as_the_defaults` 中的 id。

| text_to_image | prompt*、negative_prompt、width、height、seed、count、filename_prefix（内部） | images |
| image_to_3d | image*（内部）、seed、target_faces、texture_size、remove_background、filename_prefix（内部） | model |
~~~

```python
"""End-to-end check of the MetaOri tools through a stdio MCP session against real ComfyUI + Blender.

usage (from the fork): uv run --frozen python <this file> <tool> '<json arguments>'
Each call starts a fresh server, so jobs are also checked to survive restarts.
"""
import asyncio
import json
import os
import sys
from datetime import timedelta

from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client

FORK = r"X:\Github\Claude_MCP_UE58\Tools\blender-mcp"
SCRATCH = os.path.dirname(os.path.abspath(__file__))


async def main(tool, arguments):
    env = {**os.environ, "DISABLE_TELEMETRY": "true",
           "METAORI_OUTPUT_DIR": os.path.join(SCRATCH, "metaori_out"),
           "METAORI_EXPORT_DIR": os.path.join(SCRATCH, "metaori_export")}
    server = StdioServerParameters(command="uv", args=["run", "--frozen", "--project", FORK, "mcp-for-blender"], env=env)
    async with stdio_client(server) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()
            result = await session.call_tool(tool, arguments, read_timeout_seconds=timedelta(minutes=15))
            text = "".join(getattr(item, "text", "") for item in result.content)
            print(f"isError={result.isError}")
            print(text)


asyncio.run(main(sys.argv[1], json.loads(sys.argv[2]) if len(sys.argv) > 2 else {}))
```
